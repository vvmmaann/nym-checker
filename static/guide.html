bash <<'EOF'
PID=$(pgrep -o -x nym-node); [ -n "$PID" ] || { echo "nym-node is not running here"; exit 1; }
USR=$(ps -o user= -p "$PID" | tr -d ' '); [ "$(id -u)" = 0 ] || [ "$USR" = "$(id -un)" ] || echo "(run as root or as $USR for full output)"
ARGS=$(tr '\0' ' ' < /proc/$PID/cmdline)
ID=$(echo "$ARGS" | grep -oE -- '--id[ =][^ ]+' | head -1 | sed -E 's/--id[ =]//'); ID=${ID:-default-nym-node}
HD=$(getent passwd "$USR" | cut -d: -f6)
SVC=$(grep -oE '[^/]+\.service' /proc/$PID/cgroup | tail -1)
SSHP=${SSH_CONNECTION##* }; [ -n "$SSHP" ] || SSHP=$(sshd -T 2>/dev/null | awk '/^port /{print $2; exit}')
echo "Node ID:      $ID"
echo "Service:      ${SVC%.service}"
echo "Binary path:  $(readlink -f /proc/$PID/exe)"
echo "Server IPv4:  $(curl -4 -s --max-time 5 https://ifconfig.me || hostname -I | awk '{print $1}')"
echo "SSH port:     $SSHP"
echo "Node folder:  $HD/.nym/nym-nodes/$ID (runs as $USR)"
grep -E '^(hostname|location) = ' "$HD/.nym/nym-nodes/$ID/config/config.toml"
EOF
